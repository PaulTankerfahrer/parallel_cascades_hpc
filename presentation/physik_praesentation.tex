\begin{frame}{Warum ein Minimalmodell?}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      {\small \textbf{Das Problem}
      \begin{itemize}
        \item Die erste Wand eines Fusionsreaktors ist aus \alert{Wolfram}
          und steht im Fluss von 14-MeV-Neutronen.
        \item Ein Neutron stößt ein Gitteratom an -- das \alert{PKA}. Es
          löst in Pikosekunden eine \alert{Kollisionskaskade} aus.
        \item Zurück bleiben Defektcluster: Versprödung, begrenzte
          Lebensdauer der Wand.
      \end{itemize}}
    \end{column}
    \begin{column}{0.5\textwidth}
      {\small \textbf{Warum das Statistik verlangt}
      \begin{itemize}
        \item Interessant ist nicht die einzelne Kaskade, sondern die
          \alert{Verteilung} der Clustergrößen.
        \item Eine Verteilung braucht Hunderte bis Tausende Kaskaden --
          ein \alert{Ensemble}.
        \item Volle Molekulardynamik: Millionen Atome pro Kaskade. Für
          Ensembles zu teuer.
      \end{itemize}}
    \end{column}
  \end{columns}
  \begin{alertblock}{Der Ansatz}
    {\small Nur \textbf{Massen und brechbare Federn}, zwei Dimensionen --
    eine Kaskade in Sekunden statt Stunden.
    \textbf{Kernfrage: Wie viel Statistik überlebt in Massen und Federn?}}
  \end{alertblock}
\end{frame}

\begin{frame}{Das Modell}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      Atome auf einem \alert{Dreiecksgitter} (6 Nachbarn, schubstabil).
      Zwei Wechselwirkungen:
      \begin{itemize}
        \item \textbf{Brechbare Feder} $F=K(r-L_0)$,\\
          reißt bei 15\,\% Dehnung → \alert{Defekt}.
        \item \textbf{Steile Abstoßung} $\sim (R_\text{cut}/r)^{12}$,\\
          gibt Atomen einen Stoßquerschnitt.
      \end{itemize}
      \vspace{0.6em}
      Das PKA ist einfach ein Atom mit hoher Startgeschwindigkeit
      \[ v_\text{PKA}=\sqrt{2E_\text{PKA}/m}\;\gg\; c_\text{Schall}. \]
      Eine \alert{Hyperschall}-Kaskade.
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \includegraphics[width=0.86\linewidth]{M2_force_law.png}\\
      {\scriptsize Die beiden Kräfte: Feder (blau) + Abstoßung (rot).}\\[0.4em]
      \includegraphics[width=0.42\linewidth]{modell_ruhelage.png}\\
      {\scriptsize Modell in Ruhelage: Dreiecksgitter mit Federbindungen.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ein Ergebnis, das zu schön war}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      {\footnotesize \textbf{Das alte Resultat}\\[0.2em]
      1000 Kaskaden, alle Cluster zusammengeworfen: eine Gerade über fast
      drei Größenordnungen, $F(n)\propto n^{-S}$ mit $S\approx1{,}4$.}\\[0.2em]
      \centering
      \includegraphics[width=0.50\linewidth]{powerlaw_ccdf.png}
    \end{column}
    \begin{column}{0.52\textwidth}
      {\footnotesize \textbf{Die Prüfung nach Clauset (2009)}\\[0.4em]
      \begin{tabular}{@{}ll@{}}
        \toprule
        Goodness-of-Fit & \alert{$p = 0{,}000$}\\
        vs.\ Exponential & $R = +50$ \quad (gewinnt)\\
        vs.\ Lognormal & \alert{$R = -9{,}5$} \quad (verliert)\\
        vs.\ Cutoff & \alert{$R = -45$} \quad (verliert)\\
        \bottomrule
      \end{tabular}\\[0.4em]
      Das Potenzgesetz schlägt nur die Exponentialverteilung -- die
      Mindesthürde.}
    \end{column}
  \end{columns}
  \begin{alertblock}{Wir haben das Sampling gemessen, nicht die Kaskade}
    {\footnotesize Die PKA-Energie wurde \textbf{log-uniform} gezogen, und es
    gilt $n\sim E^{2{,}9}$ ($r=0{,}97$). Dann ist $\ln n$ gleichverteilt --
    also $p(n)\sim n^{-1}$, \textbf{unabhängig von jeder Physik}. In keinem
    einzelnen Energieband taucht die 1,4 auf.}
  \end{alertblock}
\end{frame}

\begin{frame}{Das saubere Ensemble}
  {\footnotesize \alert{$3\times300$ Läufe bei fester Energie}
  ($E=1300/1800/2400$). Einschlagort, Winkel und Seed hängen nur von der
  Lauf-Nummer ab, nicht von der Energie -- ein \alert{gepaartes} Design,
  das den Energieeffekt isoliert.}
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      {\footnotesize
      \textbf{Die alte Clusterdefinition trägt nicht.}

      Schadenszone zu \alert{einem} Klumpen: 92\,\% aller verlagerten Atome

\begin{frame}{Was die Zahlen sagen -- und was nicht}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      {\footnotesize
      \begin{itemize}
        \item \textbf{$S$ hängt von der Clusterdefinition ab.} Sechs
          Konventionen, dieselben 900 Läufe: \alert{$S=1{,}23$ bis $3{,}30$}.
          Eine Zahl ohne Angabe der Definition ist bedeutungslos -- das gilt
          auch beim Lesen von MD-Literatur.
        \item \textbf{$S$ fällt monoton mit der Kaskadengröße}
          ($2{,}50\to2{,}19\to2{,}00$), der KS-Abstand fällt mit. Bei
          $250\times250$ ist \alert{nichts konvergiert}.
        \item \textbf{Sand et al.\ (2013)} ist jetzt vergleichbar -- feste
          Energie, getrennte Cluster. Der Vergleich fällt aber andersherum
          aus: 3D-Wolfram $1{,}63$, hier 2D $\approx2{,}0$, also
          \alert{darüber}. Konsistent mit der geringeren Konnektivität in
          zwei Dimensionen.
      \end{itemize}}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      {\scriptsize
      \begin{tabular}{@{}lrrr@{}}
        \toprule
        Definition & 1300 & 1800 & 2400\\
        \midrule
        $0{,}5$ / $1{,}5$ & 1,40 & 1,35 & 1,23\\
        $1{,}0$ / $1{,}5$ & 1,57 & 1,50 & 1,45\\
        $0{,}5$ / $1{,}0$ & 1,98 & 2,10 & 2,25\\
        $1{,}0$ / $1{,}0$ & 2,50 & 2,19 & 2,00\\
        $1{,}5$ / $1{,}0$ & 2,78 & 2,51 & 2,28\\
        $2{,}0$ / $1{,}0$ & 3,30 & 3,02 & 2,77\\
        \bottomrule
      \end{tabular}\\[0.3em]
      $S$ für \texttt{disp}-Schwelle / Linkradius,\\ je Energie.}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \hrule height 0.4pt
  \vspace{0.2em}
  {\scriptsize Ehrlich bleiben: Der Goodness-of-Fit-Test verwirft auch die
  feine Definition ($p=0{,}000$) -- bei $n\approx21\,600$ verwirft er
  allerdings schon $D=0{,}013$. Der brauchbare Befund ist der \emph{Trend},
  nicht der Einzelwert.}
\end{frame}

\begin{frame}{Von der Korrektur zur Bachelorarbeit}
  {\footnotesize Die offenen Enden sind das Programm:}\\[0.2em]
  {\footnotesize
  \begin{enumerate}
    \item \textbf{Konvergiert $S$?} Größere Box, höhere Energien, Ensembles
      auf \alert{Noctua 2}. Finite-Size-Effekt oder Physik?
    \item \textbf{Clusterdefinition systematisch.} $S$ als Fläche über
      Schwelle und Linkradius statt sechs Stichproben -- gibt es ein Plateau,
      das eine Konvention auszeichnet?
    \item \textbf{Wolfram-nahe Kalibrierung.} Federkonstante, Bruchdehnung
      und Abstoßung an Kohäsionsenergie und Schwellenverlagerungsenergie
      $E_d$ anpassen.
    \item \textbf{Eigene 2D-MD-Referenz.} Dieselbe Geometrie mit echtem
      Potenzial rechnen -- dann vergleicht man Modell gegen MD, statt über
      die Dimensionen hinweg.
    \item \textbf{Die Screening-These.} Reproduziert das Minimalmodell die
      Statistik, taugt es als billiger \alert{Vorfilter}: Parameterräume grob
      absuchen, teure MD nur dort einsetzen, wo es interessant wird.
  \end{enumerate}}
  \vspace{0.4em}
  \begin{center}
    \alert{\large Die Methodik steht, die Fragen sind präzise --
    \textbf{das ist die Arbeit}.}
  \end{center}
\end{frame}

\begin{frame}{Reserve: Was ein Lauf hinterlässt}
  \begin{columns}[T]
    \begin{column}{0.53\textwidth}
      Mitgeschrieben werden drei Dinge:
      \begin{itemize}
        \item \textbf{Momentaufnahmen} während des Laufs: Ort,
          Geschwindigkeit, Verschiebung und gerissene Bindungen jedes
          Atoms -- daraus entstehen die Videos.
        \item \textbf{Energiebilanz}: kinetisch, Feder, Abstoßung,
          Bruch. Reine Kontrolle -- die Summe muss erhalten bleiben.
        \item \textbf{Endzustand} nach dem Abkühlen: \alert{eine Zeile
          pro Atom}. Aus dieser Tabelle kommt die gesamte Statistik.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \centering
      {\footnotesize Endzustand, $250\times250=62\,500$ Zeilen:}\\[0.4em]
      {\scriptsize
      \begin{tabular}{rrrr}
        \toprule
        \texttt{x} & \texttt{y} & \texttt{disp} & \texttt{broken}\\
        \midrule
        0.0000 & 0.0000 & 0.0000 & 4\\
        1.0000 & 0.0000 & 0.0000 & 2\\
        \multicolumn{4}{c}{$\vdots$}\\
        114.3486 & 83.5237 & 0.5040 & 0\\
        123.5459 & 98.3829 & 1.2109 & 2\\
        \bottomrule
      \end{tabular}}\\[0.7em]
      \begin{minipage}{0.95\linewidth}\raggedright\footnotesize
      \textbf{disp}: bleibende Auslenkung vom \emph{eigenen} Gitterplatz,
      in Gitterabständen $L_0$.\\[0.3em]
      \textbf{broken}: wie viele der sechs Bindungen gerissen sind.
      \end{minipage}
    \end{column}
  \end{columns}
\end{frame}
